local function DoRejoin()
    if isRejoining then return end
    if not CFG.AutoRejoin then return end
    if tick()-rejoinCooldown<15 then return end
    isRejoining=false; rejoinCooldown=0; task.spawn(DoRejoin)

    isTeleporting=false
    local ok,data=pcall(function() if isfile and isfile(VIP_SAVE) then return HttpService:JSONDecode(readfile(VIP_SAVE)) end end)
    local savedJobId=(ok and data and data.jobId) or jobId
    local savedPlace=(ok and data and data.placeId) or placeId
    for _=1,5 do
        local success,err=pcall(function() TeleportService:TeleportToPlaceInstance(savedPlace,savedJobId,LP) end)
        if success then isTeleporting=true; return end
        local e=tostring(err):lower()
        if e:find("teleporting") or e:find("previous") or e:find("pending") then isTeleporting=false; task.wait(20)
        else task.wait(10) end
    end
    isRejoining=false; rejoinCooldown=0; isTeleporting=false
end

local function FindNearestMob()
    if not HRP then return nil end
    local best,bestDist=nil,math.huge
    local npcs=workspace:FindFirstChild("NPCs") or workspace
    local mobLow=CFG.MobName:lower()
    local isBossMode=PORTALS[portalIndex] and PORTALS[portalIndex].IsBossEntry
    for _,m in ipairs(npcs:GetDescendants()) do
        if m:IsA("Model") and m~=Char then
            local hum=m:FindFirstChildOfClass("Humanoid"); local root=m:FindFirstChild("HumanoidRootPart")
            if hum and hum.Health>0 and root and m.Name:lower():find(mobLow,1,true) and (isBossMode or not IsBoss(m.Name)) then
                local d=(HRP.Position-root.Position).Magnitude
                if d<bestDist then bestDist=d; best=m end
            end
        end
    end
    return best
end

local MOB_GATHER_RADIUS=600
local function GetMobCentroid()
    if not HRP then return nil end
    local npcs=workspace:FindFirstChild("NPCs") or workspace
    local mobLow=CFG.MobName:lower()
    local isBossMode=PORTALS[portalIndex] and PORTALS[portalIndex].IsBossEntry
    local sum,count=Vector3.zero,0
    for _,m in ipairs(npcs:GetDescendants()) do
        if m:IsA("Model") and m~=Char then
            local hum=m:FindFirstChildOfClass("Humanoid"); local root=m:FindFirstChild("HumanoidRootPart")
            if hum and hum.Health>0 and root and m.Name:lower():find(mobLow,1,true)
                and (isBossMode or not IsBoss(m.Name)) and (HRP.Position-root.Position).Magnitude<=MOB_GATHER_RADIUS
            then sum=sum+root.Position; count+=1 end
        end
    end
    if count==0 then return nil end; return sum/count
end

local lastTPTime=0; local frozenCF,freezeUntil=nil,0; local lastCFrame=nil
local function TPToTarget(root)
    if not HRP or not root then return end
    local now=tick(); local delay=CFG.TPMinDelay+math.random()*CFG.TPRandExtra
    if now-lastTPTime<delay then return end
    lastTPTime=tick()
    local targetPos=GetMobCentroid() or root.Position
    local dest=CFrame.new(targetPos+Vector3.new(0,CFG.FloatY,0))
    HRP.AssemblyLinearVelocity=Vector3.zero; HRP.AssemblyAngularVelocity=Vector3.zero
    HRP.CFrame=dest; frozenCF=dest; freezeUntil=tick()+0; lastCFrame=dest; DisableCollision()
end

                while root and not root:IsA("BasePart") do root=root.Parent end
                if root then local txt=CreateFragESP(root); fragTargets[root]={prompt=v,text=txt} end
            end
        end
    end
end
local function GetClosestFragment()
    if not HRP then return nil end
    local closest,minDist=nil,math.huge
    for root,data in pairs(fragTargets) do
        if root and root.Parent then
            local d=(HRP.Position-root.Position).Magnitude
            if data.text then data.text.Text="Ancient\n"..math.floor(d).."m" end
            if d<minDist then minDist=d; closest={root=root,prompt=data.prompt} end
        else fragTargets[root]=nil end
    end
    return closest
end

local function W2Q_RunBossPhase()
    local function GetMapPieceProgress()
        local function scan(parent)
            for _,v in ipairs(parent:GetDescendants()) do
                if v:IsA("TextLabel") or v:IsA("TextButton") then
                    local t=v.Text; local tl=t:lower()
                    if tl:find("lost") or tl:find("progress") or tl:find("fragment") then continue end
                    local cur,total=t:match("(%d+)/(%d+)")
                    if cur and total and tonumber(total)==7 then return tonumber(cur),tonumber(total) end
                end
            end; return nil,nil
        end; local c,t=scan(LP.PlayerGui); if c then return c,t end; return scan(CoreGui)
    end
    local spawnTimer=0; local SPAWN_CD=3; local bossIdx=1; local lastPieces=-1
    while W2Q_Running do
        task.wait(0.1); RefreshChar()
        if not HRP or not Hum or Hum.Health<=0 then task.wait(1); continue end
        local cur,total=GetMapPieceProgress()
        if cur and total then
            if cur~=lastPieces then
                lastPieces=cur
                if cur>=total then W2Q_StopAndClose("✅ เควสโลก 2 เสร็จสมบูรณ์!"); return end
                bossIdx=math.min(bossIdx+1,#W2QuestBosses); spawnTimer=0
            end
        end
        local bd=W2QuestBosses[((bossIdx-1)%#W2QuestBosses)+1]
        local boss=W2Q_FindBoss(bd)
        if not boss then
            pcall(function() PortalRmt:FireServer(bd.Portal) end)
            task.wait(2); RefreshChar()
            if tick()-spawnTimer>SPAWN_CD then W2Q_SpawnBoss(bd); spawnTimer=tick() end
            task.wait(0.5)
        else
            local hum=boss:FindFirstChildOfClass("Humanoid")
            if hum and hum.Health<=0 then task.wait(0.5) else W2Q_HitBoss(boss) end
        end
    end
end
local function W2Q_RunAll()
    RefreshChar(); if not HRP or not Hum then return end
    pcall(function() PortalRmt:FireServer("World") end)
    task.wait(5); RefreshChar()
    local function findNPC()
        for _,c in ipairs({workspace:FindFirstChild("ServiceNPCs"),workspace:FindFirstChild("NPCs"),workspace}) do
            if not c then continue end
            for _,m in ipairs(c:GetDescendants()) do
                if m:IsA("Model") and m.Name=="MapQuestNPC" then return m end
            end
        end
    end
    local npcWait=0
    while not findNPC() and npcWait<5 do task.wait(0.5); npcWait+=0.5; RefreshChar() end
    local npc=findNPC()
    if npc then
        local root=npc:FindFirstChild("HumanoidRootPart")
        local pos=root and root.Position or npc:GetPivot().Position
        if HRP then HRP.CFrame=CFrame.new(pos+Vector3.new(0,2,3)) end
        task.wait(0.5)
        local function questUIVisible()
            local function scan(parent)
                for _,v in ipairs(parent:GetDescendants()) do
                    if (v:IsA("TextLabel") or v:IsA("TextButton")) and v.Visible then
                        local t=v.Text:lower()
                        if t:find("already") or t:find("past this stage") then return true end
                        if t:find("lost fragments") or t:find("map reconstruction") then return true end
                        if t:find("cartographer") and t:find("find") then return true end
                    end
                end; return false
            end; return scan(LP.PlayerGui) or scan(CoreGui)
        end
        for attempt=1,10 do
            if HRP then HRP.CFrame=CFrame.new(pos+Vector3.new(0,2,3)) end
            for _,v in ipairs(npc:GetDescendants()) do
                if v:IsA("ProximityPrompt") then v.HoldDuration=0; fireproximityprompt(v) end
            end
            task.wait(1); if questUIVisible() then break end
        end
        local msg=W2Q_CheckQuestMsg()
        if msg=="done" then W2Q_StopAndClose("เควสทำไปแล้วทั้งหมด — ปิดอัตโนมัติ"); return
        elseif msg=="next" then
            Fluent:Notify({Title="Auto W2 Quest",Content="เควสแรกผ่านแล้ว → เริ่มฆ่าบอส!",Duration=3})
            if W2Q_Running then W2Q_RunBossPhase() end; return
        end
    end
    if not W2Q_Running then return end
    fragmentCFG.Enabled=true
    local fragStart=tick()
    while W2Q_Running and tick()-fragStart<300 do
        if W2Q_HasMapReconstruction() then break end
        for _,mapName in ipairs(FRAGMENT_PORTALS) do
            if not W2Q_Running then break end
            if W2Q_HasMapReconstruction() then break end
            pcall(function() PortalRmt:FireServer(mapName) end)
            task.wait(1); RefreshChar(); ScanFragments()
            while W2Q_Running do
                local frag=GetClosestFragment(); if not frag then break end
                if frag.root and frag.root.Parent then
                    HRP.CFrame=frag.root.CFrame+Vector3.new(0,3,0); task.wait(0.3)
                    pcall(function() fireproximityprompt(frag.prompt) end)
                    task.wait(0.5); ScanFragments()
                    if W2Q_HasMapReconstruction() then break end
                else fragTargets[frag.root]=nil; break end
            end
        end
        if not W2Q_HasMapReconstruction() then
            ScanFragments()
            if not GetClosestFragment() then task.wait(2) end
        end
    end
    fragmentCFG.Enabled=false; ClearFragESP()
    if not W2Q_Running then return end
    if not W2Q_HasMapReconstruction() then W2Q_StopAndClose("⚠ ไม่พบ Map Reconstruction — หยุดอัตโนมัติ"); return end
    W2Q_RunBossPhase()
end

-- ── สแกนเฉพาะ NPCs folder เท่านั้น ไม่ตีผู้เล่น ──
local function GetDungeonMobs()
    local mobs = {}
    local playerChars = GetAllPlayerChars()

    local npcFolder = workspace:FindFirstChild("NPCs")
    local scanTarget = npcFolder or workspace

    for _, obj in ipairs(scanTarget:GetDescendants()) do
        if not obj:IsA("Model") then continue end
        if playerChars[obj] then continue end          -- ข้ามตัวละครผู้เล่นทุกคน
        if obj.Name == "CrystalModel" then continue end
        local h = obj:FindFirstChildOfClass("Humanoid")
        local r = obj:FindFirstChild("HumanoidRootPart")
        if h and h.Health > 0 and r then
            table.insert(mobs, {model=obj, hum=h, root=r})
        end
    end
    return mobs
end

local function GetCrystalPos()
    for _, obj in ipairs(workspace:GetDescendants()) do
        if obj.Name == "CrystalModel" then
            local r = obj:FindFirstChild("HumanoidRootPart") or obj:FindFirstChildOfClass("BasePart")
            if r then return r.Position + Vector3.new(0,5,0) end
        end
    end
    return nil
end

local function HitDungeonMob(hrp, mob)
    pcall(function()
        hrp.CFrame = CFrame.new(mob.root.Position + Vector3.new(0,2,0))
        task.wait(0.1)
        CombatRmt:FireServer()
        firetouchinterest(hrp, mob.root, 0)
        firetouchinterest(hrp, mob.root, 1)
        for _, part in ipairs(mob.model:GetDescendants()) do
            if part:IsA("BasePart") then
                firetouchinterest(hrp, part, 0)
                firetouchinterest(hrp, part, 1)
            end
        end
    end)
end

        end

            hrp.CFrame = cp and CFrame.new(cp) or origin
        end)
    end

local USABLE_FUNCTION = nil
task.defer(function()
    local scripts = LP:WaitForChild("PlayerScripts")
    local ls = scripts:FindFirstChildOfClass("LocalScript")
    while not ls do scripts.ChildAdded:Wait(); ls = scripts:FindFirstChildOfClass("LocalScript") end
    if getsenv then

            USABLE_FUNCTION = env._G.OpenUsableItemConfirm
        end
    end

local function HideUI(ui)
    if not ui then return end
    pcall(function()
        ui.DisplayOrder = -9999
        for _,v in ipairs(ui:GetDescendants()) do
            if v:IsA("Frame") or v:IsA("ImageLabel") or v:IsA("TextLabel")
                or v:IsA("TextButton") or v:IsA("ImageButton") then
                pcall(function() v.Visible = false end)
            end
        end
    end)
end

local _hookDone = false
local function SetupHooks()
    if _hookDone then return end; _hookDone = true
    if UseItemRmt then
        UseItemRmt.OnClientEvent:Connect(function(arg1, data)
            if type(arg1) ~= "string" or type(data) ~= "table" then return end
            if arg1 == "RaceRollComplete" then
                local got = data.race
                if got and RS.Race.on and RS.Race.target and got:lower():find(RS.Race.target:lower()) then
                    RS.Race.on = false
                    pcall(function() Fluent.Options["AutoRerollRace"]:SetValue(false) end)
                    Fluent:Notify({Title="Race ✅", Content="ได้ "..got.." แล้ว!", Duration=5})
                end
            end
            if arg1 == "ClanRollComplete" then
                local got = data.clan
                if got and RS.Clan.on and RS.Clan.target and got:lower():find(RS.Clan.target:lower()) then
                    RS.Clan.on = false
                    pcall(function() Fluent.Options["AutoRerollClan"]:SetValue(false) end)
                    Fluent:Notify({Title="Clan ✅", Content="ได้ "..got.." แล้ว!", Duration=5})
                end
            end
        end)
    end
    if TraitDataUpd then
        TraitDataUpd.OnClientEvent:Connect(function(data)
            if type(data) ~= "table" then return end
            local trait = data.RolledTrait or data.Trait
            if trait and RS.Trait.on and RS.Trait.target and trait:lower():find(RS.Trait.target:lower()) then
                RS.Trait.on = false
                pcall(function() Fluent.Options["AutoRerollTrait"]:SetValue(false) end)
                Fluent:Notify({Title="Trait ✅", Content="ได้ "..trait.." แล้ว!", Duration=5})
            end
        end)
    end
end
SetupHooks()

local function HandleConfirmUI(targetOption)
    local cui = LP.PlayerGui:FindFirstChild("ConfirmUI")
    if not cui or not cui.Enabled then return end
    local mf = cui:FindFirstChild("MainFrame"); if not mf then return end
    local txt = mf:FindFirstChild("InfoHolder") and mf.InfoHolder:FindFirstChild("Txt")
    local btns = mf:FindFirstChild("ButtonsHolder")
    if not txt or not btns then return end
    HideUI(cui)
    if txt.Text:find("Are you sure you want to") then
        pcall(function() firesignal(btns.Yes.MouseButton1Click) end); return
    end
    if txt.Text:find("Reroll your") then
        local target = RS[targetOption] and RS[targetOption].target or ""
        if txt.Text:find(target) then
            pcall(function() firesignal(btns.No.MouseButton1Click) end)
        else
            pcall(function() firesignal(btns.Yes.MouseButton1Click) end)
        end
    end
end

LP.PlayerGui.ChildAdded:Connect(function(child)
    if not (RS.Race.on or RS.Clan.on) then return end
    if child.Name ~= "ConfirmUI" then return end
    task.defer(function() HideUI(child) end)
end)

local raceCD=0; local clanCD=0; local traitCD=0; local CD=0.6

task.spawn(function()
    while task.wait(0.05) do
        local now = tick()
        if RS.Race.on and now >= raceCD then
            local cui = LP.PlayerGui:FindFirstChild("ConfirmUI")
            if cui and cui.Enabled then HandleConfirmUI("Race"); raceCD=now+CD
            elseif USABLE_FUNCTION then pcall(USABLE_FUNCTION,"Race Reroll"); raceCD=now+CD
            else pcall(function() UseItemRmt:FireServer("Use","Race Reroll",1,false,false) end); raceCD=now+CD end
        end
        if RS.Clan.on and now >= clanCD then
            local cui = LP.PlayerGui:FindFirstChild("ConfirmUI")
            if cui and cui.Enabled then HandleConfirmUI("Clan"); clanCD=now+CD
            elseif USABLE_FUNCTION then pcall(USABLE_FUNCTION,"Clan Reroll"); clanCD=now+CD
            else pcall(function() UseItemRmt:FireServer("Use","Clan Reroll",1,false,false) end); clanCD=now+CD end
        end
        if RS.Trait.on and TraitRerollRmt and now >= traitCD then
            traitCD=now+CD
            local rf,tt
            pcall(function()
                rf=TraitRerollUI.MainFrame.Frame.Content.AreYouSureYouWantToRerollFrame
                tt=TraitRerollUI.MainFrame.Frame.Content.TraitPage.TraitGottenFrame.Holder.Trait.TraitGotten
            end)
            local rfVisible=rf and rf.Visible
            local got=tt and tt.Text or ""
            HideUI(TraitRerollUI)
            if rfVisible then
                if got~="" and got:lower():find(RS.Trait.target:lower()) then
                    pcall(function() firesignal(rf.Buttons.Cancel.MouseButton1Click) end)
                else
                    pcall(function() firesignal(rf.Buttons.Accept.MouseButton1Click) end)
                end
            else
                pcall(function() TraitRerollRmt:FireServer() end)
            end
        end
    end
end)

    if not pcall(function() gui.Parent=CoreGui end) then gui.Parent=playerGui end
    local logo=Instance.new("ImageButton")
    logo.Size=UDim2.new(0,64,0,64); logo.Position=UDim2.new(0,16,0.5,-32)
    logo.Image="rbxassetid://99482842325963"; logo.BackgroundColor3=Color3.fromRGB(15,15,15)
    logo.BorderSizePixel=0; logo.ZIndex=999999; logo.Visible=true; logo.Parent=gui
    Instance.new("UICorner",logo).CornerRadius=UDim.new(0.2,0)
    local fluentSG=nil
    for i=1,30 do fluentSG=CoreGui:FindFirstChild("ScreenGui"); if fluentSG then break end; task.wait(0.3) end
    if not fluentSG then warn("[Logo] ไม่เจอ Fluent"); return end
    task.wait(1)
    local function killDropdowns()
        for _,c in ipairs(fluentSG:GetChildren()) do
            pcall(function()
                if (c:IsA("Frame") or c:IsA("ScrollingFrame")) and c.Visible then
                    local s=c.AbsoluteSize; if not(s.X>=500 and s.Y>=400) then c.Visible=false end
                end
            end)
        end
    end
    killDropdowns()
    local uiOpen=true
    local function toggleUI()
        uiOpen=not uiOpen
        for _,c in ipairs(fluentSG:GetChildren()) do
            pcall(function()
                if uiOpen then local s=c.AbsoluteSize; if(s.X>=500 and s.Y>=400) then c.Visible=true end
                else c.Visible=false end
            end)
        end
    end
    local lastToggle=0
    logo.MouseButton1Click:Connect(function()
        local now=tick(); if now-lastToggle<0.5 then return end; lastToggle=now; toggleUI()
    end)
    local dragging,dragStart,startPos=false,Vector2.new(),logo.Position
    logo.InputBegan:Connect(function(inp)
        if inp.UserInputType==Enum.UserInputType.MouseButton1 or inp.UserInputType==Enum.UserInputType.Touch then
            dragging=true; dragStart=Vector2.new(inp.Position.X,inp.Position.Y); startPos=logo.Position
        end
    end)
    UserInputSvc.InputChanged:Connect(function(inp)
        if not dragging then return end
        if inp.UserInputType==Enum.UserInputType.MouseMovement or inp.UserInputType==Enum.UserInputType.Touch then
            local dx=inp.Position.X-dragStart.X; local dy=inp.Position.Y-dragStart.Y
            logo.Position=UDim2.new(startPos.X.Scale,startPos.X.Offset+dx,startPos.Y.Scale,startPos.Y.Offset+dy)
        end
    end)
    UserInputSvc.InputEnded:Connect(function(inp)
        if inp.UserInputType==Enum.UserInputType.MouseButton1 or inp.UserInputType==Enum.UserInputType.Touch then
            dragging=false
        end
    end)
end)
